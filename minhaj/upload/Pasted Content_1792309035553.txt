  master.connect(slowFilter);slowFilter.connect(AC.destination);
  musicGain=AC.createGain();musicGain.gain.value=.4;musicGain.connect(master);
  noiseBuf=AC.createBuffer(1,AC.sampleRate,AC.sampleRate);

function beep(f,dur,type,vol,slideTo){
  if(!AC||muted)return;
  type=type||'square';vol=vol===undefined?.15:vol;
  var o=AC.createOscillator();o.type=type;
  o.frequency.setValueAtTime(f,AC.currentTime);
  if(slideTo)o.frequency.exponentialRampToValueAtTime(Math.max(20,slideTo),AC.currentTime+dur);
  var g=AC.createGain();g.gain.setValueAtTime(vol,AC.currentTime);
  g.gain.exponentialRampToValueAtTime(.0001,AC.currentTime+dur);
  o.connect(g);g.connect(master);o.start();o.stop(AC.currentTime+dur+.02);
}

}

}

function tickMusic(){
  if(!AC||muted)return;
  var spb=60/112/2; /* croches à 112 BPM */
  if(musicNext<AC.currentTime-.5)musicNext=AC.currentTime+.1;
  while(musicNext<AC.currentTime+.3){
    var i=musicStep,t=musicNext;
    var meas=(i>>3)%8,pos=i&7;
    if(pos===0)mNote(BASS[meas],t,1.5,'sawtooth',.11,340);
    if(pos===0||pos===4)mKick(t);
    if(pos%2===1)mHat(t);
    if((i&1)===0){
      var idx=(meas*3+(pos>>1)*2)%PENTA.length;
      mNote(PENTA[idx]*(rageV>.5&&pos===6?2:1),t,.17,'square',0.02,2600);
    }
    musicStep=(musicStep+1)%64;
    musicNext+=spb;
  }
}

var tileMeshes=new Map(),tilePool=[],stores=new Map(),opLog=[],dirtyTiles=new Set(),texUploads=[];
var OPLOGCAP=4000,logCapWarned=false;
function storeFor(tx,ty){
  var k=tx+','+ty,st=stores.get(k);
  if(!st){
    if(stores.size>3000){
      var removed=false;
      stores.forEach(function(v,kk){if(!removed&&!tileMeshes.has(kk)){stores.delete(kk);removed=true;}});
      if(!removed){var first=null;stores.forEach(function(v,kk){if(first===null)first=kk;});if(first!==null)stores.delete(first);}
    }
    st={ops:[],score:0};stores.set(k,st);
  }
  return st;
}
function newTile(){
  var c=mk();c.width=T;c.height=T;
  return {c:c,x:c.getContext('2d'),tx:0,ty:0,key:'',tex:null,texReady:false,m:new Float32Array(16)};
}
function assignTile(tile,tx,ty){
  tile.tx=tx;tile.ty=ty;tile.key=tx+','+ty;
  var c=tile.x;
  c.setTransform(1,0,0,1,0,0);
  c.clearRect(0,0,T,T);
  drawDecor(c,tx,ty);
  var st=storeFor(tx,ty);
  for(var i=0;i<st.ops.length;i++)drawOp(c,st.ops[i],tx*T,ty*T);
  m4compose(tile.m,tx*T+T/2,0,ty*T+T/2,0,0,0,T,1,T);
  tile.texReady=false;
  if(texUploads.indexOf(tile)<0)texUploads.push(tile);
}
function updateTiles(rect){
  var m=200;
  var tx0=Math.floor((rect.x0-m)/T),tx1=Math.floor((rect.x1+m)/T);
  var ty0=Math.floor((rect.y0-m)/T),ty1=Math.floor((rect.y1+m)/T);
  var needed=new Set();
  for(var ty=ty0;ty<=ty1;ty++)for(var tx=tx0;tx<=tx1;tx++){
    var k=tx+','+ty;needed.add(k);
    if(!tileMeshes.has(k)){
      var tile=tilePool.pop()||newTile();
      if(!tile.tex)tile.tex=gl.createTexture();
      assignTile(tile,tx,ty);
      tileMeshes.set(k,tile);
    }
  }
  tileMeshes.forEach(function(tile,k){
    if(!needed.has(k)){tilePool.push(tile);tileMeshes.delete(k);}
  });
}
function processDirty(){
  if(!dirtyTiles.size)return;
  var n=0;
  dirtyTiles.forEach(function(k){
    if(n>=4)return;
    dirtyTiles.delete(k);
    var tile=tileMeshes.get(k);
    if(tile){var p=k.split(',');assignTile(tile,+p[0],+p[1]);}
    n++;
  });
}
function uploadDirtyTextures(){
  var n=0;
  while(texUploads.length&&n<4){
    var tile=texUploads.shift();
    if(!tileMeshes.has(tile.key)&&tilePool.indexOf(tile)<0)continue;
    if(!tile.tex)tile.tex=gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D,tile.tex);
    gl.texImage2D(gl.TEXTURE_2D,0,gl.RGBA,gl.RGBA,gl.UNSIGNED_BYTE,tile.c);
    gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MIN_FILTER,gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MAG_FILTER,gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_S,gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_T,gl.CLAMP_TO_EDGE);
    tile.texReady=true;
    n++;
  }
}
function genCracks(x,y,a,len,out){
  var pts=[[x,y]],n=ri(3,5),st=len/n,px=x,py=y,ca=a,i;
  for(i=0;i<n;i++){ca+=rnd(-.35,.35);px+=Math.cos(ca)*st;py+=Math.sin(ca)*st;pts.push([px,py]);}
  out.push({pts:pts,w:rnd(1.8,2.8),al:.5+rnd(.15)});
  if(Math.random()<.5)genCracks(px,py,ca+rnd(-.9,.9),len*.5,out);
}

function opScore(op){
  if(op.t==='g')return op.d*1.1;
  if(op.t==='m')return op.d*1.5;
  if(op.t==='s')return op.r*op.r*.05;
  if(op.t==='o')return op.r*op.r*.8;
  if(op.t==='h')return 150;
  if(op.t==='b')return 26;
  return 3;
}
function applyOp(op){
  var x0,x1,y0,y1;
  if(op.t==='b'){x0=op.x-70;x1=op.x+70;y0=op.y-70;y1=op.y+70;}
  else if(op.t==='h'){x0=op.x-160;x1=op.x+160;y0=op.y-160;y1=op.y+160;}
  else if(op.t==='g'||op.t==='m'){
    x0=Math.min(op.x1,op.x2)-24;x1=Math.max(op.x1,op.x2)+24;
    y0=Math.min(op.y1,op.y2)-24;y1=Math.max(op.y1,op.y2)+24;
  }
  else if(op.t==='s'){x0=op.x-op.r-4;x1=op.x+op.r+4;y0=op.y-op.r-4;y1=op.y+op.r+4;}
  else if(op.t==='o'){x0=op.x-op.r*2-8;x1=op.x+op.r*2+8;y0=op.y-op.r*2-8;y1=op.y+op.r*2+8;}
  else{x0=op.x-10;x1=op.x+10;y0=op.y-10;y1=op.y+10;}
  var tx0=Math.floor(x0/T),tx1=Math.floor(x1/T),ty0=Math.floor(y0/T),ty1=Math.floor(y1/T);
  var tiles=[];
  for(var ty=ty0;ty<=ty1;ty++)for(var tx=tx0;tx<=tx1;tx++){
    var k=tx+','+ty;
    var st=storeFor(tx,ty);
    st.ops.push(op);st.score+=opScore(op);
    if(st.ops.length>900)st.ops.shift();
    var tile=tileMeshes.get(k);
    if(tile){
      drawOp(tile.x,op,tx*T,ty*T);
      if(texUploads.indexOf(tile)<0)texUploads.push(tile);
      tile.texReady=false;
    }
    tiles.push(k);
  }
  if(opLog.length<OPLOGCAP)opLog.push({op:op,tiles:tiles});
  else if(!logCapWarned){logCapWarned=true;
    showToast('La machine à remonter le temps atteint sa limite : le passé lointain est figé.');}
}

function rewindStep(){
  if(!opLog.length){finishRewind();return;}
  var chunk=Math.min(14,opLog.length),i,j;
  for(i=0;i<chunk;i++){
    var e=opLog.pop();
    for(j=0;j<e.tiles.length;j++){
      var k=e.tiles[j];
      var st=stores.get(k);if(!st)continue;
      var idx=st.ops.indexOf(e.op);
      if(idx>=0)st.ops.splice(idx,1);
      var s=0;for(var m=0;m<st.ops.length;m++)s+=opScore(st.ops[m]);
      st.score=s;
      dirtyTiles.add(k);
    }
    stats.rewound++;
    var ox=(e.op.x!==undefined)?e.op.x:e.op.x1;
    var oz=(e.op.y!==undefined)?e.op.y:e.op.y1;
    if(ox!==undefined&&Math.random()<.25)
      glow(ox+rnd(-4,4),6,oz+rnd(-4,4),0,90,0,rnd(.4,.7),'t',3);
  }
  if(now-lastRwS>80){
    lastRwS=now;
    beep(220+(1-opLog.length/Math.max(1,OPLOGCAP))*800,.05,'sawtooth',.07);
  }
  if(!opLog.length)finishRewind();
}

/* ============================ matrices 4x4 ============================ */
function m4ident(o){o=o||new Float32Array(16);o[0]=1;o[1]=0;o[2]=0;o[3]=0;o[4]=0;o[5]=1;o[6]=0;o[7]=0;
  o[8]=0;o[9]=0;o[10]=1;o[11]=0;o[12]=0;o[13]=0;o[14]=0;o[15]=1;return o;}
var IDENT=m4ident();
function m4mul(out,a,b){
  var a0=a[0],a1=a[1],a2=a[2],a3=a[3],a4=a[4],a5=a[5],a6=a[6],a7=a[7],
      a8=a[8],a9=a[9],a10=a[10],a11=a[11],a12=a[12],a13=a[13],a14=a[14],a15=a[15];
  for(var i=0;i<4;i++){
    var b0=b[i*4],b1=b[i*4+1],b2=b[i*4+2],b3=b[i*4+3];
    out[i*4]  =b0*a0+b1*a4+b2*a8 +b3*a12;
    out[i*4+1]=b0*a1+b1*a5+b2*a9 +b3*a13;
    out[i*4+2]=b0*a2+b1*a6+b2*a10+b3*a14;
    out[i*4+3]=b0*a3+b1*a7+b2*a11+b3*a15;
  }
  return out;
}
function m4compose(o,px,py,pz,rx,ry,rz,sx,sy,sz){
  var cy=Math.cos(ry),sy_=Math.sin(ry),cx=Math.cos(rx),sx_=Math.sin(rx),cz=Math.cos(rz),sz_=Math.sin(rz);
  var m00=cy*cz+sy_*sx_*sz_, m01=-cy*sz_+sy_*sx_*cz_, m02=sy_*cx;
  var m10=cx*sz_, m11=cx*cz_, m12=-sx_;
  var m20=-sy_*cz+cy*sx_*sz_, m21=sy_*sz_+cy*sx_*cz_, m22=cy*cx;
  o[0]=m00*sx;o[1]=m10*sx;o[2]=m20*sx;o[3]=0;
  o[4]=m01*sy;o[5]=m11*sy;o[6]=m21*sy;o[7]=0;
  o[8]=m02*sz;o[9]=m12*sz;o[10]=m22*sz;o[11]=0;
  o[12]=px;o[13]=py;o[14]=pz;o[15]=1;
  return o;
}
function m4persp(o,fovy,aspect,near,far){
  var f=1/Math.tan(fovy/2),nf=1/(near-far);
  o[0]=f/aspect;o[1]=0;o[2]=0;o[3]=0;
  o[4]=0;o[5]=f;o[6]=0;o[7]=0;
  o[8]=0;o[9]=0;o[10]=(far+near)*nf;o[11]=-1;
  o[12]=0;o[13]=0;o[14]=2*far*near*nf;o[15]=0;
  return o;
}
function m4ortho(o,l,r,b,t,n,f){
  o[0]=2/(r-l);o[1]=0;o[2]=0;o[3]=0;
  o[4]=0;o[5]=2/(t-b);o[6]=0;o[7]=0;
  o[8]=0;o[9]=0;o[10]=-2/(f-n);o[11]=0;
  o[12]=-(r+l)/(r-l);o[13]=-(t+b)/(t-b);o[14]=-(f+n)/(f-n);o[15]=1;
  return o;
}
function m4lookAt(o,ex,ey,ez,cx,cy,cz,ux,uy,uz){
  var zx=ex-cx,zy=ey-cy,zz=ez-cz;
  var xx=uy*zz-uz*zy,xy=uz*zx-ux*zz,xz=ux*zy-uy*zx;
  var len=Math.hypot(xx,xy,xz);
  if(len<1e-6){xx=1;xy=0;xz=0;}else{xx/=len;xy/=len;xz/=len;}
  var yx=zy*xz-zz*xy,yy=zz*xx-zx*xz,yz=zx*xy-zy*xx;
  o[0]=xx;o[1]=yx;o[2]=zx;o[3]=0;
  o[4]=xy;o[5]=yy;o[6]=zy;o[7]=0;
  o[8]=xz;o[9]=yz;o[10]=zz;o[11]=0;
  o[12]=-(xx*ex+xy*ey+xz*ez);
  o[13]=-(yx*ex+yy*ey+yz*ez);
  o[14]=-(zx*ex+zy*ey+zz*ez);
  o[15]=1;
  return o;
}
function m4rotZ(o,a){
  var c=Math.cos(a),s=Math.sin(a);
  o[0]=c;o[1]=s;o[2]=0;o[3]=0;
  o[4]=-s;o[5]=c;o[6]=0;o[7]=0;
  o[8]=0;o[9]=0;o[10]=1;o[11]=0;
  o[12]=0;o[13]=0;o[14]=0;o[15]=1;
  return o;
}
function m4invert(out,a){
  var a00=a[0],a01=a[1],a02=a[2],a03=a[3],
      a10=a[4],a11=a[5],a12=a[6],a13=a[7],
      a20=a[8],a21=a[9],a22=a[10],a23=a[11],
      a30=a[12],a31=a[13],a32=a[14],a33=a[15],
      b00=a00*a11-a01*a10,b01=a00*a12-a02*a10,b02=a00*a13-a03*a10,
      b03=a01*a12-a02*a11,b04=a01*a13-a03*a11,b05=a02*a13-a03*a12,
      b06=a20*a31-a21*a30,b07=a20*a32-a22*a30,b08=a20*a33-a23*a30,
      b09=a21*a32-a22*a31,b10=a21*a33-a23*a31,b11=a22*a33-a23*a32,
      det=b00*b11-b01*b10+b02*b09+b03*b08-b04*b07+b05*b06;
  if(!det)return null;
  det=1/det;
  out[0]=(a11*b11-a12*b10+a13*b09)*det;
  out[1]=(a02*b10-a01*b11-a03*b09)*det;
  out[2]=(a31*b05-a32*b04+a33*b03)*det;
  out[3]=(a22*b04-a21*b05-a23*b03)*det;
  out[4]=(a12*b08-a10*b11-a13*b07)*det;
  out[5]=(a00*b11-a02*b08+a03*b07)*det;
  out[6]=(a32*b02-a30*b05-a33*b01)*det;
  out[7]=(a20*b05-a22*b02+a23*b01)*det;
  out[8]=(a10*b10-a11*b08+a13*b06)*det;
  out[9]=(a01*b08-a00*b10-a03*b06)*det;
  out[10]=(a30*b04-a31*b02+a33*b00)*det;
  out[11]=(a21*b02-a20*b04-a23*b00)*det;
  out[12]=(a11*b07-a10*b09-a12*b06)*det;
  out[13]=(a00*b09-a01*b07+a02*b06)*det;
  out[14]=(a31*b01-a30*b03-a32*b00)*det;
  out[15]=(a20*b03-a21*b01+a22*b00)*det;
  return out;
}
function mulVec4(m,v){

    m[3]*v[0]+m[7]*v[1]+m[11]*v[2]+m[15]*v[3]
  ];
}

function sh(type,src){
  var s=gl.createShader(type);
  gl.shaderSource(s,src);gl.compileShader(s);
  if(!gl.getShaderParameter(s,gl.COMPILE_STATUS))
    throw new Error('Shader: '+(gl.getShaderInfoLog(s)||'?'));
  return s;
}
function progInfo(p){
  var info={p:p,u:{},a:{}};
  var nu=gl.getProgramParameter(p,gl.ACTIVE_UNIFORMS);
  for(var i=0;i<nu;i++){var inf=gl.getActiveUniform(p,i);
    info.u[inf.name.replace('[0]','')]=gl.getUniformLocation(p,inf.name);}
  var na=gl.getProgramParameter(p,gl.ACTIVE_ATTRIBUTES);
  for(var j=0;j<na;j++){var ia=gl.getActiveAttrib(p,j);
    info.a[ia.name]=gl.getAttribLocation(p,ia.name);}
  return info;
}
function mkProg(vs,fs){
  var p=gl.createProgram();
  gl.attachShader(p,sh(gl.VERTEX_SHADER,vs));
  gl.attachShader(p,sh(gl.FRAGMENT_SHADER,fs));
  gl.linkProgram(p);
  if(!gl.getProgramParameter(p,gl.LINK_STATUS))
    throw new Error('Link: '+(gl.getProgramInfoLog(p)||'?'));
  return progInfo(p);
}

function mkRT(w,h){
  var fbo=gl.createFramebuffer();
  var tex=gl.createTexture();
  gl.bindTexture(gl.TEXTURE_2D,tex);
  gl.texImage2D(gl.TEXTURE_2D,0,gl.RGBA,w,h,0,gl.RGBA,gl.UNSIGNED_BYTE,null);
  gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MIN_FILTER,gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MAG_FILTER,gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_S,gl.CLAMP_TO_EDGE);
  gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_T,gl.CLAMP_TO_EDGE);
  var rb=gl.createRenderbuffer();
  gl.bindRenderbuffer(gl.RENDERBUFFER,rb);
  gl.renderbufferStorage(gl.RENDERBUFFER,gl.DEPTH_COMPONENT16,w,h);
  gl.bindFramebuffer(gl.FRAMEBUFFER,fbo);
  gl.framebufferTexture2D(gl.FRAMEBUFFER,gl.COLOR_ATTACHMENT0,gl.TEXTURE_2D,tex,0);
  gl.framebufferRenderbuffer(gl.FRAMEBUFFER,gl.DEPTH_ATTACHMENT,gl.RENDERBUFFER,rb);
  var ok=gl.checkFramebufferStatus(gl.FRAMEBUFFER)===gl.FRAMEBUFFER_COMPLETE;
  gl.bindFramebuffer(gl.FRAMEBUFFER,null);
  return ok?{fbo:fbo,tex:tex,w:w,h:h}:null;
}

function ndcFromEvent(e){
  var r=cv.getBoundingClientRect();
  return [((e.clientX-r.left)/r.width)*2-1, -(((e.clientY-r.top)/r.height)*2-1)];
}
function ndcToGround(e){
  if(!engineOK)return null;
  var nd=ndcFromEvent(e);
  var p0=mulVec4(INVVP,[nd[0],nd[1],-1,1]);
  var p1=mulVec4(INVVP,[nd[0],nd[1],1,1]);
  if(Math.abs(p0[3])<1e-6||Math.abs(p1[3])<1e-6)return null;
  var a=[p0[0]/p0[3],p0[1]/p0[3],p0[2]/p0[3]];
  var b=[p1[0]/p1[3],p1[1]/p1[3],p1[2]/p1[3]];
  var dy=b[1]-a[1];
  if(Math.abs(dy)<1e-6)return null;
  var t=-a[1]/dy;
  if(t<0)return null;
  return {x:a[0]+(b[0]-a[0])*t,z:a[2]+(b[2]-a[2])*t};
}

function groundRectFromCamera(){
  var xs=[],zs=[];
  var cs=[[-1,-1],[1,-1],[-1,1],[1,1]];
  for(var i=0;i<4;i++){
    var p0=mulVec4(INVVP,[cs[i][0],cs[i][1],-1,1]);
    var p1=mulVec4(INVVP,[cs[i][0],cs[i][1],1,1]);
    if(Math.abs(p0[3])<1e-6||Math.abs(p1[3])<1e-6)continue;
    var a=[p0[0]/p0[3],p0[1]/p0[3],p0[2]/p0[3]];
    var b=[p1[0]/p1[3],p1[1]/p1[3],p1[2]/p1[3]];
    var dy=b[1]-a[1];
    if(Math.abs(dy)<1e-6)continue;
    var t=-a[1]/dy;
    if(t<0)continue;
    xs.push(a[0]+(b[0]-a[0])*t);
    zs.push(a[2]+(b[2]-a[2])*t);
  }
  if(xs.length<4)return {x0:camTX-600,x1:camTX+600,y0:camTZ-500,y1:camTZ+500};
  return {x0:Math.min.apply(null,xs),x1:Math.max.apply(null,xs),
          y0:Math.min.apply(null,zs),y1:Math.max.apply(null,zs)};
}

  BOUNDS={x0:r.x0+30,x1:r.x1-30,z0:r.y0+30,z1:r.y1-30};
}

  var az=tod*TAU,sh=Math.sin(az);
  var day=sstep(-.06,.28,sh);
  nightV=1-day;
  if(sh>0){var t=sstep(0,.35,sh);lerp3(G_skyTop,duskTop,dayTop,t);lerp3(G_skyHor,duskHor,dayHor,t);}
  else{var t2=sstep(0,.22,-sh);lerp3(G_skyTop,duskTop,nightTop,t2);lerp3(G_skyHor,duskHor,nightHor,t2);}
  G_fog[0]=G_skyHor[0];G_fog[1]=G_skyHor[1];G_fog[2]=G_skyHor[2];
  var ldx=Math.cos(az),ldy=Math.max(.18,sh),ldz=.45;

  var ll=Math.hypot(ldx,ldy,ldz);ldx/=ll;ldy/=ll;ldz/=ll;
  G_ldir[0]=ldx;G_ldir[1]=ldy;G_ldir[2]=ldz;
  var warm=[1,.77,.55],white=[1,.96,.9];
  lerp3(G_lcol,warm,white,sstep(0,.4,sh));
  var li=.22+1.25*Math.max(0,sh);
  G_lcol[0]*=li;G_lcol[1]*=li;G_lcol[2]*=li;
  var ambD=[.4,.42,.46],ambN=[.16,.19,.3];
  lerp3(G_amb,ambN,ambD,day);

function m4basisBeam(out,x1,y1,z1,x2,y2,z2,th,len){
  var dx=x2-x1,dy=y2-y1,dz=z2-z1;
  var l=Math.hypot(dx,dy,dz)||1;
  dx/=l;dy/=l;dz/=l;
  var ux=0,uy=1,uz=0;
  if(Math.abs(dy)>.98){ux=1;uy=0;uz=0;}
  var xx=uy*dz-uz*dy,xy=uz*dx-ux*dz,xz=ux*dy-uy*dx;
  var xl=Math.hypot(xx,xy,xz)||1;xx/=xl;xy/=xl;xz/=xl;
  var yx=dy*xz-dz*xy,yy=dz*xx-dx*xz,yz=dx*xy-dy*xx;
  out[0]=xx*th;out[1]=xy*th;out[2]=xz*th;out[3]=0;
  out[4]=yx*th;out[5]=yy*th;out[6]=yz*th;out[7]=0;
  out[8]=dx*len;out[9]=dy*len;out[10]=dz*len;out[11]=0;
  out[12]=(x1+x2)/2;out[13]=(y1+y2)/2;out[14]=(z1+z2)/2;out[15]=1;
  return out;
}

function postChain(){
  if(!rtScene||!rtBright||!rtA||!rtB)throw new Error('RT manquant');
  gl.disable(gl.DEPTH_TEST);
  gl.depthMask(false);
  /* LA passe critique : la scène est rendue AVANT d'être composée */
  gl.bindFramebuffer(gl.FRAMEBUFFER,rtScene.fbo);
  drawScene(rtScene.w,rtScene.h);
  /* bright */
  gl.bindFramebuffer(gl.FRAMEBUFFER,rtBright.fbo);
  gl.viewport(0,0,rtBright.w,rtBright.h);
  gl.useProgram(prBright.p);
  gl.activeTexture(gl.TEXTURE0);
  gl.bindTexture(gl.TEXTURE_2D,rtScene.tex);
  gl.uniform1i(prBright.u.tD,0);
  gl.uniform1f(prBright.u.uTh,.6);
  gl.uniform1f(prBright.u.uBoost,1.55);
  fullscreenPass(prBright);
  /* blur H + V (2 itérations = halo doux) */
  gl.useProgram(prBlur.p);
  gl.uniform1i(prBlur.u.tD,0);
  for(var it=0;it<2;it++){
    gl.bindFramebuffer(gl.FRAMEBUFFER,rtA.fbo);
    gl.viewport(0,0,rtA.w,rtA.h);
    gl.bindTexture(gl.TEXTURE_2D,rtBright.tex);
    gl.uniform2f(prBlur.u.dir,1,0);
    gl.uniform2f(prBlur.u.res,rtA.w,rtA.h);
    fullscreenPass(prBlur);
    gl.bindFramebuffer(gl.FRAMEBUFFER,rtB.fbo);
    gl.viewport(0,0,rtB.w,rtB.h);
    gl.bindTexture(gl.TEXTURE_2D,rtA.tex);
    gl.uniform2f(prBlur.u.dir,0,1);
    gl.uniform2f(prBlur.u.res,rtB.w,rtB.h);
    fullscreenPass(prBlur);
    if(it===0){
      gl.bindFramebuffer(gl.FRAMEBUFFER,rtBright.fbo);
      gl.viewport(0,0,rtBright.w,rtBright.h);
      gl.bindTexture(gl.TEXTURE_2D,rtB.tex);
      gl.uniform2f(prBlur.u.dir,.7,.7);
      fullscreenPass(prBlur);
    }
  }
  /* composite → écran */
  gl.bindFramebuffer(gl.FRAMEBUFFER,null);
  gl.viewport(0,0,cv.width,cv.height);
  gl.useProgram(prComp.p);
  gl.activeTexture(gl.TEXTURE0);
  gl.bindTexture(gl.TEXTURE_2D,rtScene.tex);
  gl.activeTexture(gl.TEXTURE1);
  gl.bindTexture(gl.TEXTURE_2D,rtB.tex);
  gl.uniform1i(prComp.u.tD,0);
  gl.uniform1i(prComp.u.tB,1);
  gl.uniform1f(prComp.u.uT,now*.001);
  gl.uniform1f(prComp.u.uRage,rageV);
  gl.uniform1f(prComp.u.uRewind,rwdV);
  gl.uniform1f(prComp.u.uNight,nightV);
  gl.uniform1f(prComp.u.uSlow,slowVis);
  gl.uniform1f(prComp.u.uAspect,W/Math.max(1,H));
  fullscreenPass(prComp);
  gl.activeTexture(gl.TEXTURE0);
  gl.depthMask(true);
  gl.enable(gl.DEPTH_TEST);
}

}

function drawHUD(dt){

  hctx.clearRect(0,0,W,H);
  var i;
  for(i=popups.length-1;i>=0;i--){
    var p=popups[i];
    p.age+=dt;
    if(p.age>=p.life){popups.splice(i,1);continue;}
  }
  for(i=0;i<popups.length;i++){
    var pp=popups[i];
    var s=worldToScreen(pp.x,26+pp.age*60,pp.z);
    if(!s)continue;
    var q=pp.age/pp.life;
    var sc=(pp.big?1.5:1)*(q<.12?.5+q/.12*.5:1);
    hctx.save();
    hctx.translate(s.x,s.y);
    hctx.scale(sc,sc);
    hctx.globalAlpha=q>.7?(1-q)/.3:1;
    hctx.font='bold '+(pp.big?22:16)+'px '+FONT_S;
    hctx.textAlign='center';
    hctx.lineWidth=4;hctx.strokeStyle='rgba(0,0,0,.75)';
    hctx.strokeText(pp.txt,0,0);
    hctx.fillStyle=pp.big?'#ff8b1e':'#ffd23e';
    hctx.fillText(pp.txt,0,0);
    hctx.restore();
  }
  /* réticules des pointeurs actifs */
  pointers.forEach(function(p){
    if(!p.down)return;
    var s=worldToScreen(p.x,2,p.z);
    if(!s)return;
    var col=TOOLCOL[tool]||'#fff';
    hctx.save();
    hctx.translate(s.x,s.y);
    hctx.rotate(now*.003);
    hctx.strokeStyle=col;hctx.lineWidth=2;
    hctx.beginPath();hctx.arc(0,0,13,0,TAU);hctx.stroke();
    hctx.rotate(-now*.003);
    hctx.beginPath();
    hctx.moveTo(-19,0);hctx.lineTo(-9,0);hctx.moveTo(9,0);hctx.lineTo(19,0);
    hctx.moveTo(0,-19);hctx.lineTo(0,-9);hctx.moveTo(0,9);hctx.lineTo(0,19);
    hctx.stroke();
    hctx.restore();
  });
}

var last=performance.now(),errShown=0;
function frame(t){
  now=t;
  var dtRaw=Math.min(.05,(t-last)/1000);last=t;
  var target=1;
  if(hitT>0){hitT-=dtRaw;target=.06;}
  else if(slowT>0){slowT-=dtRaw;target=.35;}
  timeScale+=(target-timeScale)*Math.min(1,dtRaw*20);
  var dt=dtRaw*timeScale;
  if(slowFilter){
    var f=20000-((1-timeScale)*16500);
    try{slowFilter.frequency.setTargetAtTime(Math.max(500,f),AC.currentTime,.08);}catch(e){}
  }
  try{step(dt,dtRaw);}
  catch(err){
    console.error('[Destructo3D]',err);
    if(now-errShown>5000){errShown=now;showToast('Un bug a survécu. Il a été neutralisé.');}
  }
  requestAnimationFrame(frame);
}
function step(dt,dtRaw){
  if(state==='fin'||state==='finale')return;
  if(!engineOK)return;
  tod=(tod+dt/DAYLEN)%1;
  flashV*=Math.exp(-7*dtRaw);
  camKick*=Math.exp(-9*dtRaw);
  if(state==='play'){
    heroUpdate(dt);
    pointers.forEach(function(p){if(p.down)toolTick(p,dt);});
    tickBullets(dt);
    if(mode==='infini'){
      camTX+=(hero.x-camTX)*Math.min(1,dtRaw*3.2);
      camTZ+=(hero.z-camTZ)*Math.min(1,dtRaw*3.2);
    }else{
      camTX+=(0-camTX)*Math.min(1,dtRaw*8);
      camTZ+=(0-camTZ)*Math.min(1,dtRaw*8);
    }
  }else if(state==='repair'){
    repT+=dtRaw;
    if(repT>=.75){state='play';setPlaying(true);showToast('Peinture fraîche. Ne pas toucher pendant 24 h.');}
  }else if(state==='rewind'){
    rewindStep();
  }
  updateCombo(dt);
  animateHero(dt);
  if(state!=='rewind'){tickTermis(dt);tickDebris(dt);}
  tickGlow(dt);tickSmoke(dt);tickRings(dt);tickDust(dt);
  shake*=Math.exp(-6*dtRaw);if(shake<.01)shake=0;
  var sx=shake>.2?rnd(-shake,shake):0,sz=shake>.2?rnd(-shake,shake):0;
  placeCamera(sx,sz);
  updateLighting();
  updateShadowCam();
  updateTiles(groundRectFromCamera());
  processDirty();
  uploadDirtyTextures();
  renderShadow();
  renderFrame();
  slowVis+=(((timeScale<.7)?1:0)-slowVis)*Math.min(1,dtRaw*8);
  var rageTarget=(state==='play')?clamp((pct-55)/45,0,1):0;
  rageV+=(rageTarget-rageV)*Math.min(1,dtRaw*2);
  if(rageV>.55&&!seenRage){seenRage=true;showToast('LA RAGE DÉFORME LA RÉALITÉ (5e dimension).');sfx.rage();slowT=Math.max(slowT,1);}
  if(rageV<.3)seenRage=false;
  rwdV+=((state==='rewind'?1:0)-rwdV)*Math.min(1,dtRaw*8);
  drawHUD(dt);
  tickMusic();
  pctT+=dtRaw;if(pctT>=.5){pctT=0;samplePct();}
}
